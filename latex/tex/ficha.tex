\newcount\fichaNP \fichaNP=0

% --- autores: "Nome Sobrenome, Nome Sobrenome" ---
\newcount\fichaI
\newcommand{\fichaAutoresDiretos}{\fichaI=1 \fichaAutoresLoop}
\def\fichaAutoresLoop{\ifnum\fichaI>\numAutores\else
  \ifnum\fichaI>1 , \fi
  \csname autor\the\fichaI n\endcsname\ \csname autor\the\fichaI s\endcsname
  \advance\fichaI by 1 \expandafter\fichaAutoresLoop\fi}
% --- assuntos: "1. a. 2. b." ---
\newcommand{\fichaAssuntos}{\fichaI=1 \fichaAssuntosLoop}
\def\fichaAssuntosLoop{\ifnum\fichaI>\fichaNP\else
  \ifnum\fichaI>1 \ \fi
  \the\fichaI. \csname fichaP\the\fichaI\endcsname.%
  \advance\fichaI by 1 \expandafter\fichaAssuntosLoop\fi}
% --- entradas secundárias: "I. Sobrenome, Nome. II. ..." (coautores, orientador, título) ---
\newcounter{fichaE}
\newcommand{\fichaEntrada}[1]{\stepcounter{fichaE}\Roman{fichaE}. #1}
\newcommand{\fichaEntradasAutores}{\fichaI=2 \fichaEntradasLoop}
\def\fichaEntradasLoop{\ifnum\fichaI>\numAutores\else
  \fichaEntrada{\csname autor\the\fichaI s\endcsname, \csname autor\the\fichaI n\endcsname.} %
  \advance\fichaI by 1 \expandafter\fichaEntradasLoop\fi}
